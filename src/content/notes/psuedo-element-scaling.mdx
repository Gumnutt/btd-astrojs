---
title: "Scaling elements with pseudo-elements on hover"
description: "A simple way to scale elements using pseudo-elements on hover."
published: 2025-05-27
---

import Demo from "@components/demos/psuedo-element-scale.astro"

<Demo />

```css
.scale-background-on-hover {
  position: relative;
  isolation: isolate;
}
.scale-background-on-hover::after {
  background-color: pink;
  content: "";
  position: absolute;
  top: 0;
  left: 0;
  right: 0;
  bottom: 0;
  z-index: -1;
  transition: transform 250ms;
}
.scale-background-on-hover:hover::after {
  transform: scale(1.1);
}
```
